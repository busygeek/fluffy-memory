package com.mockapi.controller;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.mockapi.exception.MockApiException;
import com.mockapi.model.ApiResponse;
import com.mockapi.model.RouteConfig;
import com.mockapi.service.DynamicRouteService;
import com.mockapi.service.FileStorageService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import jakarta.servlet.http.HttpServletRequest;
import java.util.*;

@RestController
@RequestMapping("/api/dynamic")
@RequiredArgsConstructor
@Slf4j
public class DynamicEndpointController {
    
    private final DynamicRouteService dynamicRouteService;
    private final FileStorageService fileStorage;
    private final ObjectMapper objectMapper;
    
    @RequestMapping(value = "/**", method = {RequestMethod.GET, RequestMethod.POST, 
                   RequestMethod.PUT, RequestMethod.DELETE, RequestMethod.PATCH})
    public ResponseEntity<ApiResponse> handleDynamicRequest(
            HttpServletRequest request,
            @RequestBody(required = false) Map<String, Object> requestBody) {
        
        String httpMethod = request.getMethod();
        String requestPath = "/api/dynamic" + request.getServletPath().substring("/api/dynamic".length());
        
        log.debug("Processing dynamic request: {} {}", httpMethod, requestPath);
        
        // Find matching route configuration
        RouteConfig route = dynamicRouteService.findMatchingRoute(httpMethod, requestPath);
        if (route == null) {
            throw new MockApiException(404, "No dynamic route found for " + httpMethod + " " + requestPath);
        }
        
        log.debug("Found matching route: {}", route.getRouteId());
        
        // Extract path variables and query parameters
        Map<String, String> pathVars = dynamicRouteService.extractPathVariables(route, requestPath);
        Map<String, String> queryParams = extractQueryParameters(request);
        
        // Process the request based on HTTP method
        Object result = switch (httpMethod.toUpperCase()) {
            case "GET" -> handleGetRequest(route, pathVars, queryParams);
            case "POST" -> handlePostRequest(route, pathVars, queryParams, requestBody);
            case "PUT" -> handlePutRequest(route, pathVars, queryParams, requestBody);
            case "DELETE" -> handleDeleteRequest(route, pathVars, queryParams);
            case "PATCH" -> handlePatchRequest(route, pathVars, queryParams, requestBody);
            default -> throw new MockApiException(405, "Method not allowed: " + httpMethod);
        };
        
        String message = String.format("Dynamic %s request processed successfully", httpMethod);
        Map<String, Object> metadata = Map.of(
            "route", route.getRouteId(),
            "collection", route.getCollection(),
            "pathVariables", pathVars,
            "queryParameters", queryParams
        );
        
        return ResponseEntity.ok(ApiResponse.custom(200, message, result, metadata));
    }
    
    // FIXED GET REQUEST HANDLER - Now supports both filtering approaches
    private Object handleGetRequest(RouteConfig route, Map<String, String> pathVars, 
                                   Map<String, String> queryParams) {
        String collection = route.getCollection();
        
        // If there's an 'id' path variable, try to get specific item
        if (pathVars.containsKey("id")) {
            String id = pathVars.get("id");
            Object item = getItemFromCollection(collection, id, route.getStorageStrategy());
            if (item == null) {
                throw new MockApiException(404, "Item with id " + id + " not found in " + collection);
            }
            return item;
        }
        
        // Get all items from collection
        List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
            new TypeReference<List<Map<String, Object>>>() {});
        
        List<Map<String, Object>> filteredItems;
        Map<String, Object> filterInfo;
        
        // Choose filtering approach based on whether paramMappings exist
        if (route.getParamMappings() != null && !route.getParamMappings().isEmpty()) {
            log.debug("Using mapped parameter filtering for route: {}", route.getRouteId());
            
            // Apply parameter mappings and use mapped filtering
            Map<String, Object> mappedFilters = new HashMap<>();
            applyParameterMappings(route, pathVars, queryParams, mappedFilters);
            
            filteredItems = applyMappedFilters(items, mappedFilters);
            
            filterInfo = Map.of(
                "filterType", "mapped",
                "originalParams", queryParams,
                "mappedFilters", mappedFilters,
                "paramMappings", route.getParamMappings()
            );
        } else {
            log.debug("Using direct parameter filtering for route: {}", route.getRouteId());
            
            // Use direct filtering (current approach)
            filteredItems = applyDirectFilters(items, queryParams);
            
            filterInfo = Map.of(
                "filterType", "direct",
                "appliedFilters", queryParams
            );
        }
        
        return Map.of(
            "collection", collection,
            "items", filteredItems,
            "count", filteredItems.size(),
            "totalCount", items.size(),
            "filterInfo", filterInfo
        );
    }
    
    // ENHANCED DIRECT FILTERING: Now supports array bracket syntax and nested fields
    private List<Map<String, Object>> applyDirectFilters(List<Map<String, Object>> items, 
                                                         Map<String, String> queryParams) {
        if (queryParams.isEmpty()) {
            log.debug("No query parameters provided, returning all items");
            return items;
        }
        
        log.debug("Applying direct filters: {}", queryParams);
        
        return items.stream()
                .filter(item -> {
                    boolean matches = queryParams.entrySet().stream()
                            .allMatch(entry -> {
                                String paramName = entry.getKey();      // Direct query param name (can include array syntax)
                                String paramValue = entry.getValue();   // Query param value
                                
                                Object itemValue;
                                
                                // Check if the parameter name contains array brackets or dots (nested access)
                                if (paramName.contains("[]") || paramName.contains(".")) {
                                    // Use nested field access for complex field paths
                                    itemValue = getNestedFieldValue(item, paramName);
                                    log.debug("Direct nested/array filter: field='{}', resolved value='{}'", paramName, itemValue);
                                } else {
                                    // Simple field access for flat fields
                                    itemValue = item.get(paramName);
                                    log.debug("Direct simple filter: field='{}', value='{}'", paramName, itemValue);
                                }
                                
                                if (itemValue == null) {
                                    log.debug("Field '{}' not found in item {}", paramName, item.get("id"));
                                    return false;
                                }
                                
                                boolean containsMatch = matchFieldValue(itemValue, paramValue);
                                
                                log.debug("Direct filter check: field='{}', itemValue='{}', paramValue='{}', match={}",
                                        paramName, itemValue, paramValue, containsMatch);
                                
                                return containsMatch;
                            });
                    
                    log.debug("Item {} matches all direct filters: {}", item.get("id"), matches);
                    return matches;
                })
                .collect(ArrayList::new, ArrayList::add, ArrayList::addAll);
    }
    
    // NEW APPROACH: Mapped filtering with improved array bracket syntax
    private List<Map<String, Object>> applyMappedFilters(List<Map<String, Object>> items, 
                                                         Map<String, Object> mappedFilters) {
        if (mappedFilters.isEmpty()) {
            log.debug("No mapped filters provided, returning all items");
            return items;
        }
        
        log.debug("Applying mapped filters: {}", mappedFilters);
        
        return items.stream()
                .filter(item -> {
                    boolean matches = mappedFilters.entrySet().stream()
                            .allMatch(entry -> {
                                String mappedFieldName = entry.getKey();        // Mapped field name
                                String filterValue = String.valueOf(entry.getValue()); // Filter value
                                
                                // Handle nested field access with array bracket syntax
                                Object itemValue = getNestedFieldValue(item, mappedFieldName);
                                
                                if (itemValue == null) {
                                    log.debug("Mapped field '{}' not found in item {}", mappedFieldName, item.get("id"));
                                    return false;
                                }
                                
                                boolean containsMatch = matchFieldValue(itemValue, filterValue);
                                
                                log.debug("Mapped filter check: field='{}', itemValue='{}', filterValue='{}', match={}",
                                        mappedFieldName, itemValue, filterValue, containsMatch);
                                
                                return containsMatch;
                            });
                    
                    log.debug("Item {} matches all mapped filters: {}", item.get("id"), matches);
                    return matches;
                })
                .collect(ArrayList::new, ArrayList::add, ArrayList::addAll);
    }
    
    // Enhanced field value matching with array and nested object support
    private boolean matchFieldValue(Object itemValue, String filterValue) {
        if (itemValue == null) {
            return false;
        }
        
        // Handle arrays (e.g., skills: ["Java", "Python"])
        if (itemValue instanceof List<?> list) {
            return list.stream()
                    .anyMatch(element -> element != null && 
                             element.toString().toLowerCase().contains(filterValue.toLowerCase()));
        }
        
        // Handle regular fields
        return itemValue.toString().toLowerCase().contains(filterValue.toLowerCase());
    }
    
    // Enhanced nested field access with array bracket syntax support
    // Supports: "user.profile.name", "projects[].status", "projects[].tags[]"
    private Object getNestedFieldValue(Map<String, Object> item, String fieldPath) {
        if (fieldPath == null || fieldPath.isEmpty()) {
            return null;
        }
        
        log.debug("Getting nested field value for path: '{}'", fieldPath);
        
        // Split the field path while preserving array brackets
        String[] pathParts = splitFieldPath(fieldPath);
        Object currentValue = item;
        
        for (String part : pathParts) {
            if (currentValue == null) {
                log.debug("Current value is null at part: '{}'", part);
                return null;
            }
            
            // Check if this part has array bracket syntax
            if (part.endsWith("[]")) {
                // Remove the [] to get the field name
                String fieldName = part.substring(0, part.length() - 2);
                log.debug("Processing array field: '{}'", fieldName);
                
                if (currentValue instanceof Map<?, ?> map) {
                    Object arrayField = map.get(fieldName);
                    if (arrayField instanceof List<?> list) {
                        // Return the list itself - matchFieldValue will handle array searching
                        currentValue = list;
                        log.debug("Found array field '{}' with {} elements", fieldName, list.size());
                    } else {
                        log.debug("Field '{}' is not an array", fieldName);
                        return null;
                    }
                } else {
                    log.debug("Current value is not a map when looking for array field '{}'", fieldName);
                    return null;
                }
            } else {
                // Regular field access
                if (currentValue instanceof Map<?, ?> map) {
                    currentValue = map.get(part);
                    log.debug("Accessed field '{}', new value: {}", part, currentValue);
                } else if (currentValue instanceof List<?> list) {
                    // If we have a list and a regular field name, search in all list elements
                    log.debug("Searching for field '{}' in list elements", part);
                    List<Object> foundValues = new ArrayList<>();
                    
                    for (Object element : list) {
                        if (element instanceof Map<?, ?> elementMap) {
                            Object fieldValue = elementMap.get(part);
                            if (fieldValue != null) {
                                foundValues.add(fieldValue);
                            }
                        }
                    }
                    
                    if (foundValues.isEmpty()) {
                        log.debug("Field '{}' not found in any list elements", part);
                        return null;
                    } else if (foundValues.size() == 1) {
                        currentValue = foundValues.get(0);
                        log.debug("Found single value for field '{}': {}", part, currentValue);
                    } else {
                        currentValue = foundValues;
                        log.debug("Found multiple values for field '{}': {}", part, foundValues);
                    }
                } else {
                    log.debug("Cannot access field '{}' on non-map/non-list value: {}", part, currentValue);
                    return null;
                }
            }
        }
        
        log.debug("Final nested field value: {}", currentValue);
        return currentValue;
    }
    
    // Split field path while preserving array brackets
    // "projects[].status" -> ["projects[]", "status"]
    // "user.profile.name" -> ["user", "profile", "name"]
    // "projects[].tags[]" -> ["projects[]", "tags[]"]
    private String[] splitFieldPath(String fieldPath) {
        List<String> parts = new ArrayList<>();
        StringBuilder currentPart = new StringBuilder();
        
        for (int i = 0; i < fieldPath.length(); i++) {
            char c = fieldPath.charAt(i);
            
            if (c == '.') {
                // Add current part and start new one
                if (currentPart.length() > 0) {
                    parts.add(currentPart.toString());
                    currentPart = new StringBuilder();
                }
            } else {
                currentPart.append(c);
            }
        }
        
        // Add the last part
        if (currentPart.length() > 0) {
            parts.add(currentPart.toString());
        }
        
        String[] result = parts.toArray(new String[0]);
        log.debug("Split field path '{}' into parts: {}", fieldPath, Arrays.toString(result));
        return result;
    }
    
    // Apply parameter mappings (existing method, now used for GET as well)
    private void applyParameterMappings(RouteConfig route, Map<String, String> pathVars, 
                                       Map<String, String> queryParams, Map<String, Object> data) {
        if (route.getParamMappings() != null) {
            route.getParamMappings().forEach((paramName, dataKey) -> {
                // Try to get value from path variables first, then query parameters
                String value = pathVars.getOrDefault(paramName, queryParams.get(paramName));
                if (value != null) {
                    data.put(dataKey, value);
                    log.debug("Applied parameter mapping: '{}' -> '{}' = '{}'", paramName, dataKey, value);
                }
            });
        }
    }
    
    // Rest of the existing methods remain unchanged...
    
    private Object handlePostRequest(RouteConfig route, Map<String, String> pathVars, 
                                    Map<String, String> queryParams, Map<String, Object> requestBody) {
        String collection = route.getCollection();
        
        if (requestBody == null) {
            requestBody = new HashMap<>();
        }
        
        // Apply path variables to request body if mapping exists
        applyParameterMappings(route, pathVars, queryParams, requestBody);
        
        // Generate ID if not provided
        if (!requestBody.containsKey("id")) {
            requestBody.put("id", UUID.randomUUID().toString());
        }
        
        requestBody.put("createdAt", new Date());
        
        String itemId = String.valueOf(requestBody.get("id"));
        boolean useIndividualFile = "individual_file".equals(route.getStorageStrategy());
        
        if (useIndividualFile) {
            fileStorage.writeItemToCollection(collection, itemId, requestBody);
        } else {
            List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
                new TypeReference<List<Map<String, Object>>>() {});
            items.add(requestBody);
            fileStorage.writeCollectionData(collection, items);
        }
        
        return requestBody;
    }
    
    private Object handlePutRequest(RouteConfig route, Map<String, String> pathVars, 
                                   Map<String, String> queryParams, Map<String, Object> requestBody) {
        String collection = route.getCollection();
        
        if (!pathVars.containsKey("id")) {
            throw new MockApiException(400, "ID path variable required for PUT requests");
        }
        
        String id = pathVars.get("id");
        
        if (requestBody == null) {
            requestBody = new HashMap<>();
        }
        
        requestBody.put("id", id);
        requestBody.put("updatedAt", new Date());
        
        applyParameterMappings(route, pathVars, queryParams, requestBody);
        
        boolean useIndividualFile = "individual_file".equals(route.getStorageStrategy());
        
        if (useIndividualFile) {
            fileStorage.writeItemToCollection(collection, id, requestBody);
        } else {
            updateItemInCollection(collection, id, requestBody);
        }
        
        return requestBody;
    }
    
    private Object handleDeleteRequest(RouteConfig route, Map<String, String> pathVars, 
                                      Map<String, String> queryParams) {
        String collection = route.getCollection();
        
        if (!pathVars.containsKey("id")) {
            throw new MockApiException(400, "ID path variable required for DELETE requests");
        }
        
        String id = pathVars.get("id");
        boolean useIndividualFile = "individual_file".equals(route.getStorageStrategy());
        
        boolean deleted;
        if (useIndividualFile) {
            deleted = fileStorage.deleteItemFromCollection(collection, id);
        } else {
            deleted = deleteItemFromCollection(collection, id);
        }
        
        if (!deleted) {
            throw new MockApiException(404, "Item not found: " + id);
        }
        
        return Map.of("deleted", true, "id", id);
    }
    
    private Object handlePatchRequest(RouteConfig route, Map<String, String> pathVars, 
                                     Map<String, String> queryParams, Map<String, Object> requestBody) {
        // For PATCH, we merge the request body with existing data
        String collection = route.getCollection();
        
        if (!pathVars.containsKey("id")) {
            throw new MockApiException(400, "ID path variable required for PATCH requests");
        }
        
        String id = pathVars.get("id");
        
        // Get existing item
        Object existing = getItemFromCollection(collection, id, route.getStorageStrategy());
        if (existing == null) {
            throw new MockApiException(404, "Item not found: " + id);
        }
        
        @SuppressWarnings("unchecked")
        Map<String, Object> existingMap = (Map<String, Object>) existing;
        
        // Merge with request body
        if (requestBody != null) {
            existingMap.putAll(requestBody);
        }
        
        existingMap.put("updatedAt", new Date());
        applyParameterMappings(route, pathVars, queryParams, existingMap);
        
        boolean useIndividualFile = "individual_file".equals(route.getStorageStrategy());
        
        if (useIndividualFile) {
            fileStorage.writeItemToCollection(collection, id, existingMap);
        } else {
            updateItemInCollection(collection, id, existingMap);
        }
        
        return existingMap;
    }
    
    // Helper methods remain the same...
    
    private Object getItemFromCollection(String collection, String id, String storageStrategy) {
        if ("individual_file".equals(storageStrategy)) {
            return fileStorage.readItemFromCollection(collection, id, Map.class);
        } else {
            List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
                new TypeReference<List<Map<String, Object>>>() {});
            
            return items.stream()
                    .filter(item -> id.equals(String.valueOf(item.get("id"))))
                    .findFirst()
                    .orElse(null);
        }
    }
    
    private void updateItemInCollection(String collection, String id, Map<String, Object> updatedItem) {
        List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
            new TypeReference<List<Map<String, Object>>>() {});
        
        for (int i = 0; i < items.size(); i++) {
            Map<String, Object> item = items.get(i);
            if (id.equals(String.valueOf(item.get("id")))) {
                items.set(i, updatedItem);
                fileStorage.writeCollectionData(collection, items);
                return;
            }
        }
        
        throw new MockApiException(404, "Item not found: " + id);
    }
    
    private boolean deleteItemFromCollection(String collection, String id) {
        List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
            new TypeReference<List<Map<String, Object>>>() {});
        
        boolean removed = items.removeIf(item -> id.equals(String.valueOf(item.get("id"))));
        
        if (removed) {
            fileStorage.writeCollectionData(collection, items);
        }
        
        return removed;
    }
    
    private Map<String, String> extractQueryParameters(HttpServletRequest request) {
        Map<String, String> queryParams = new HashMap<>();
        if (request.getQueryString() != null) {
            String[] pairs = request.getQueryString().split("&");
            for (String pair : pairs) {
                String[] keyValue = pair.split("=");
                if (keyValue.length == 2) {
                    try {
                        queryParams.put(
                            java.net.URLDecoder.decode(keyValue[0], "UTF-8"),
                            java.net.URLDecoder.decode(keyValue[1], "UTF-8")
                        );
                    } catch (Exception e) {
                        log.warn("Failed to decode query parameter: {}", pair);
                    }
                }
            }
        }
        return queryParams;
    }
}
