// CHANGE 2: Enhanced TemplateProcessorService to store generated values
// ========================================

@Service
@RequiredArgsConstructor
@Slf4j
public class TemplateProcessorService {
    
    private final ObjectMapper objectMapper;
    private static final Pattern TEMPLATE_PATTERN = Pattern.compile("\\{\\{([^}]+)\\}\\}");
    
    // ... existing methods ...
    
    private String resolveVariable(String variableName, TemplateContext context) {
        String resolvedValue = switch (variableName) {
            // Generated IDs - CHANGED to store in context
            case "created_record_guid", "generated_guid" -> {
                String guid = UUID.randomUUID().toString();
                storeGeneratedValue(context, variableName, guid);
                yield guid;
            }
            case "generated_6_digit_id" -> {
                String sixDigitId = String.format("%06d", new Random().nextInt(999999));
                storeGeneratedValue(context, variableName, sixDigitId);
                yield sixDigitId;
            }
            
            // ... rest of existing cases unchanged ...
            
            default -> {
                // ... existing default logic ...
            }
        };
        
        return resolvedValue;
    }
    
    // NEW METHOD: Store generated values in context
    private void storeGeneratedValue(TemplateContext context, String variableName, String value) {
        if (context.getGeneratedValues() == null) {
            context.setGeneratedValues(new HashMap<>());
        }
        context.getGeneratedValues().put(variableName, value);
        log.debug("Stored generated value: {} = {}", variableName, value);
    }
    
    // NEW METHOD: Get generated values from context
    public Map<String, String> getGeneratedValues(TemplateContext context) {
        return context.getGeneratedValues() != null ? context.getGeneratedValues() : new HashMap<>();
    }
}

// CHANGE 3: Update Enhanced Endpoint Controller to save generated IDs
// ========================================

@RestController
@RequestMapping("/api/dynamic")
@RequiredArgsConstructor
@Slf4j
public class EnhancedDynamicEndpointController {
    
    // ... existing fields ...
    
    private Object handlePostRequest(RouteConfig route, Map<String, String> pathVars, 
                                    Map<String, String> queryParams, Map<String, Object> requestBody) {
        String collection = route.getRequestConfig().getCollection();
        
        if (requestBody == null) {
            requestBody = new HashMap<>();
        }
        
        // Apply path variables to request body if mapping exists
        applyParameterMappings(route.getRequestConfig(), pathVars, queryParams, requestBody);
        
        // Generate ID if not provided
        if (!requestBody.containsKey("id")) {
            requestBody.put("id", UUID.randomUUID().toString());
        }
        
        requestBody.put("createdAt", new Date());
        
        // NEW: Process templates first to generate IDs, then save them to record
        TemplateContext context = buildTemplateContext(requestBody, queryParams, pathVars, 
                new HttpHeaders(), requestBody);
        
        ResponseData responseData = dynamicRouteService.buildResponse(route, requestBody, 
                getRequestPath(), queryParams, new HttpHeaders(), pathVars, requestBody);
        
        // NEW: Get generated values from template processing
        Map<String, String> generatedValues = templateProcessor.getGeneratedValues(context);
        
        // NEW: Save generated values to the actual record
        if (!generatedValues.isEmpty()) {
            for (Map.Entry<String, String> entry : generatedValues.entrySet()) {
                String templateVar = entry.getKey();
                String generatedValue = entry.getValue();
                
                // Map template variables to record fields
                String recordField = mapTemplateVariableToRecordField(templateVar);
                if (recordField != null) {
                    requestBody.put(recordField, generatedValue);
                    log.debug("Saved generated value to record: {} = {}", recordField, generatedValue);
                }
            }
        }
        
        String itemId = String.valueOf(requestBody.get("id"));
        boolean useIndividualFile = "individual_file".equals(route.getRequestConfig().getStorageStrategy());
        
        // Save the record with generated IDs included
        if (useIndividualFile) {
            fileStorage.writeItemToCollection(collection, itemId, requestBody);
        } else {
            List<Map<String, Object>> items = fileStorage.readCollectionData(collection, 
                new TypeReference<List<Map<String, Object>>>() {});
            items.add(requestBody);
            fileStorage.writeCollectionData(collection, items);
        }
        
        return requestBody; // Return the record with generated IDs
    }
    
    // NEW METHOD: Map template variables to record field names
    private String mapTemplateVariableToRecordField(String templateVariable) {
        return switch (templateVariable) {
            case "created_record_guid" -> "id"; // Save main GUID as record ID
            case "generated_guid" -> "guid"; // Save additional GUID as 'guid' field
            case "generated_6_digit_id" -> "contactId"; // Save 6-digit ID as 'contactId'
            default -> null; // Don't save other template variables to record
        };
    }
    
    // ... rest of existing methods unchanged ...
}

// CHANGE 4: Enhanced Route Service to support generated ID mapping
// ========================================

@Service
@RequiredArgsConstructor
@Slf4j
public class EnhancedDynamicRouteService {
    
    // ... existing methods unchanged ...
    
    // UPDATED: Build response with generated values tracking
    public ResponseData buildResponse(RouteConfig route, Object data, String requestPath, 
                                    Map<String, String> queryParams, HttpHeaders requestHeaders,
                                    Map<String, String> pathParams, Object requestBody) {
        ResponseConfig respConfig = route.getResponseConfig();
        
        if (respConfig == null) {
            return ResponseData.builder()
                    .statusCode(200)
                    .body(data)
                    .headers(new HashMap<>())
                    .delayMs(0)
                    .build();
        }
        
        // Build template context with generated values tracking
        TemplateContext context = buildTemplateContext(data, queryParams, pathParams, requestHeaders, requestBody);
        context.setGeneratedValues(new HashMap<>()); // Initialize tracking
        
        // ... rest of method unchanged ...
        
        return ResponseData.builder()
                .statusCode(statusCode)
                .body(responseBody)
                .headers(processedHeaders)
                .delayMs(respConfig.getDelayMs())
                .build();
    }
    
    // ... rest of existing methods unchanged ...
}
