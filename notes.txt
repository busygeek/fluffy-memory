@Transactional
public void processMessages() {
    List<String> messages = new ArrayList<>();
    messageQueue.drainTo(messages, BATCH_SIZE);

    if (messages.isEmpty()) {
        return;
    }

    LinkedHashMap<String, ParentEntity> parentMap = new LinkedHashMap<>();
    List<ChildEntity1> children1ToSave = new ArrayList<>();
    List<ChildEntity2> children2ToSave = new ArrayList<>();

    // Collect unique refValues and refTypeIds from messages
    Set<String> refValues = new LinkedHashSet<>();
    Set<Integer> refTypeIds = new LinkedHashSet<>();
    for (String message : messages) {
        refValues.add(extractRefValue(message));
        refTypeIds.add(extractRefTypeId(message));
    }

    // Batch fetch existing parents by refValue and refTypeId in one query
    List<ParentEntity> existingParents = parentRepository.findByRefValueInAndRefTypeIdIn(refValues, refTypeIds);
    for (ParentEntity parent : existingParents) {
        parentMap.put(parent.getRefValue(), parent);
    }

    // Process messages in order, ensuring the latest parent data is used
    for (String message : messages) {
        String refValue = extractRefValue(message);  // refValue from message
        Integer refTypeId = extractRefTypeId(message);  // refTypeId from message
        String userId = extractUserId(message);
        Integer attributeId = extractAttributeId(message);  // Extract attributeId
        Integer statusId = extractStatusId(message);  // Extract statusId
        String value = extractValue(message);  // Extract value field
        String finalValue = extractFinalValue(message);  // Extract finalValue field

        // Get the parent if it exists, or create a new one if not
        ParentEntity parent = parentMap.get(refValue);
        if (parent == null) {
            // Create a new parent entity only if it doesn't exist
            parent = new ParentEntity();
            parent.setRefValue(refValue);  // Set refValue
            parent.setRefTypeId(refTypeId);  // Set refTypeId
            parent.setCreateTime(Instant.now());  // Set createTime only for new parents
            parentMap.put(refValue, parent); // Add the new parent to the map
        }

        // Always update the parent fields (existing or new)
        parent.setUserId(userId);
        parent.setUpdateTime(Instant.now());  // Always update the updateTime

		
		 // Check if ChildEntity1 exists
        ChildEntity1 existingChild1 = childRepository1.findByAttributeIdAndReferenceId(attributeId, parent.getReferenceId());
        if (existingChild1 != null) {
            // Update existing child1 fields (except attributeValueId, attributeId, createTime)
            existingChild1.setStatusId(statusId);
            existingChild1.setValue(value);
            existingChild1.setFinalValue(finalValue);
            existingChild1.setModifyTime(Instant.now());
            existingChild1.setUserId(userId);
            children1ToSave.add(existingChild1);  // Add to save list
        } else {
            // Create new child1 entity if not found
            ChildEntity1 child1 = new ChildEntity1();
            child1.setParent(parent);
            child1.setAttributeId(attributeId);
            child1.setReferenceId(parent.getReferenceId());  // Set referenceId from parent
            child1.setStatusId(statusId);
            child1.setValue(value);
            child1.setFinalValue(finalValue);
            child1.setCreateTime(Instant.now());
            child1.setModifyTime(Instant.now());
            child1.setUserId(userId);
            children1ToSave.add(child1);  // Add to save list
        }

        // Check if ChildEntity2 exists
        ChildEntity2 existingChild2 = childRepository2.findByAttributeIdAndReferenceId(attributeId, parent.getReferenceId());
        if (existingChild2 != null) {
            // Update existing child2 fields (except attributeValueId, attributeId, createTime)
            existingChild2.setStatusId(statusId);
            existingChild2.setValue(value);
            existingChild2.setFinalValue(finalValue);
            existingChild2.setModifyTime(Instant.now());
            existingChild2.setUserId(userId);
            children2ToSave.add(existingChild2);  // Add to save list
        } else {
            // Create new child2 entity if not found
            ChildEntity2 child2 = new ChildEntity2();
            child2.setParent(parent);
            child2.setAttributeId(attributeId);
            child2.setReferenceId(parent.getReferenceId());  // Set referenceId from parent

    }

    // Save all parents (whether new or existing)
    parentRepository.saveAll(parentMap.values());
    parentRepository.flush(); // Ensure parents are persisted and IDs are generated for new ones

    // After parents are saved, update the children with the correct parent ID
    childRepository1.saveAll(children1ToSave);
    childRepository2.saveAll(children2ToSave);
}
