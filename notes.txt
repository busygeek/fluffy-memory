// ========================================
// SIMPLE SOLUTION: Enhanced Template Processor
// ========================================

@Service
@RequiredArgsConstructor
@Slf4j
public class TemplateProcessorService {
    
    private final ObjectMapper objectMapper;
    private static final Pattern TEMPLATE_PATTERN = Pattern.compile("\\{\\{([^}]+)\\}\\}");
    
    // Simple map to store generated values during processing
    private Map<String, String> currentGeneratedValues = new HashMap<>();
    
    public Object processTemplate(Object template, TemplateContext context) {
        // Clear previous generated values
        currentGeneratedValues.clear();
        
        // Process template as before
        Object result = processTemplateInternal(template, context);
        
        // Save generated values to the record if it's a Map
        if (context.getCreatedObject() instanceof Map) {
            Map<String, Object> record = (Map<String, Object>) context.getCreatedObject();
            saveGeneratedValuesToRecord(record);
        }
        
        return result;
    }
    
    private Object processTemplateInternal(Object template, TemplateContext context) {
        if (template == null) return null;
        
        if (template instanceof String) {
            return processStringTemplate((String) template, context);
        } else if (template instanceof Map) {
            return processMapTemplate((Map<String, Object>) template, context);
        } else if (template instanceof List) {
            return processListTemplate((List<Object>) template, context);
        }
        
        return template;
    }
    
    private String resolveVariable(String variableName, TemplateContext context) {
        return switch (variableName) {
            // Generate and store these values
            case "created_record_guid", "generated_guid" -> {
                String guid = UUID.randomUUID().toString();
                currentGeneratedValues.put(variableName, guid);
                yield guid;
            }
            case "generated_6_digit_id" -> {
                String sixDigitId = String.format("%06d", new Random().nextInt(999999));
                currentGeneratedValues.put(variableName, sixDigitId);
                yield sixDigitId;
            }
            case "generated_5digit_alphanumeric" -> {
                String alphanumeric = generateAlphanumeric(5);
                currentGeneratedValues.put(variableName, alphanumeric);
                yield alphanumeric;
            }
            
            // All other template variables (unchanged)
            case "current_timestamp" -> LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME);
            case "current_date" -> LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE);
            case "timestamp" -> String.valueOf(System.currentTimeMillis());
            case "found_user" -> serializeToString(context.getFoundObject());
            case "request_body" -> serializeToString(context.getRequestBody());
            // ... all other existing cases unchanged
            
            default -> {
                if (variableName.startsWith("query.")) {
                    String paramName = variableName.substring(6);
                    yield context.getQueryParams().getOrDefault(paramName, "");
                } else if (variableName.startsWith("path.")) {
                    String paramName = variableName.substring(5);
                    yield context.getPathParams().getOrDefault(paramName, "");
                } else if (variableName.startsWith("header.")) {
                    String headerName = variableName.substring(7);
                    yield context.getHeaders().getOrDefault(headerName, "");
                } else {
                    log.warn("Unknown template variable: {}", variableName);
                    yield "{{" + variableName + "}}";
                }
            }
        };
    }
    
    // Simple method to save generated values to record
    private void saveGeneratedValuesToRecord(Map<String, Object> record) {
        for (Map.Entry<String, String> entry : currentGeneratedValues.entrySet()) {
            String templateVar = entry.getKey();
            String generatedValue = entry.getValue();
            
            // Save with simple field names
            String fieldName = switch (templateVar) {
                case "created_record_guid" -> "recordId";
                case "generated_guid" -> "guid";
                case "generated_6_digit_id" -> "contactId";
                case "generated_5digit_alphanumeric" -> "alphanumericId";
                default -> templateVar.replace("generated_", "").replace("created_record_", "");
            };
            
            record.put(fieldName, generatedValue);
            log.debug("Saved to record: {} = {}", fieldName, generatedValue);
        }
    }
    
    // Helper method for alphanumeric generation
    private String generateAlphanumeric(int length) {
        String chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
        Random random = new Random();
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < length; i++) {
            result.append(chars.charAt(random.nextInt(chars.length())));
        }
        return result.toString();
    }
    
    // ... rest of existing methods unchanged (processStringTemplate, processMapTemplate, etc.)
}

// ========================================

// Just add new cases to resolveVariable():
case "generated_8digit_hex" -> {
    String hex = String.format("%08X", new Random().nextInt());
    currentGeneratedValues.put(variableName, hex);
    yield hex;
}
case "generated_uuid_short" -> {
    String shortUuid = UUID.randomUUID().toString().substring(0, 8);
    currentGeneratedValues.put(variableName, shortUuid);
    yield shortUuid;
}

// And they'll automatically be saved to records when used in templates!
