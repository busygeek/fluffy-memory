else if (variableName.startsWith("requestbody.")) {
                    // NEW: Handle requestbody.fieldname patterns
                    String fieldName = variableName.substring(12);
                    yield getValueFromRequestBody(fieldName, context);
                } else {
                    log.warn("Unknown template variable: {}", variableName);
                    yield "{{" + variableName + "}}";
                }
            }
        };
    }
    
    // NEW METHOD: Extract values from request body
    private String getValueFromRequestBody(String fieldName, TemplateContext context) {
        if (context.getRequestBody() == null) {
            return "";
        }
        
        try {
            if (context.getRequestBody() instanceof Map) {
                Map<String, Object> body = (Map<String, Object>) context.getRequestBody();
                
                // Handle nested field paths like "user.name" or simple fields like "guid"
                String[] pathParts = fieldName.split("\\.");
                Object value = body;
                
                for (String part : pathParts) {
                    if (value instanceof Map) {
                        value = ((Map<String, Object>) value).get(part);
                    } else {
                        return "";
                    }
                    
                    if (value == null) {
                        return "";
                    }
                }
                
                return String.valueOf(value);
            }
        } catch (Exception e) {
            log.warn("Error extracting field '{}' from request body: {}", fieldName, e.getMessage());
        }
        
        return "";
    }
